# Runtime dependencies for teac-dab-controls.
# Installed into a dedicated virtualenv by install.sh.

# Volumio websocket client. Must stay on the socketio 4.x / engineio 3.x line
# to remain compatible with Volumio's socket.io server protocol.
python-engineio==3.14.2
python-socketio[client]>=4,<5

# GPIO / SPI hardware access. The MCP3008 button ADC is read directly via
# bitbanged GPIO or spidev (see includes/controls.py) - no Adafruit stack needed.
#
# RPi.GPIO and spidev are NOT listed here: both need compiling, which would
# drag python3-dev (and, on this image, a forced core-library upgrade) into
# the venv. install.sh installs them prebuilt via apt (python3-rpi.gpio,
# python3-spidev) and the venv is created with --system-site-packages so it
# can see them.
pigpio

# REST API server
fastapi
uvicorn

# Retry helper used by the Volumio client
tenacity

# LCD menu library (forked). Needs 2.2.0 or later for create_char() and
# render_frame(), which the level meter uses -- against anything older it fails
# at runtime with AttributeError rather than at install time.
git+https://github.com/domb84/rpi-lcd-menu.git@master
