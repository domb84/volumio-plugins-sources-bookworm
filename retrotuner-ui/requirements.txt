# Runtime dependencies for teac-dab-controls.
# Installed into a dedicated virtualenv by install.sh.

# Volumio websocket client. Must stay on the socketio 4.x / engineio 3.x line
# to remain compatible with Volumio's socket.io server protocol.
python-engineio==3.14.2
python-socketio[client]>=4,<5

# GPIO / SPI hardware access. The MCP3008 button ADC is read directly via
# bitbanged GPIO or spidev (see includes/controls.py) - no Adafruit stack needed.
#
# RPi.GPIO and spidev are NOT listed here: both need compiling, which would
# drag python3-dev (and, on this image, a forced core-library upgrade) into
# the venv. install.sh installs them prebuilt via apt (python3-rpi.gpio,
# python3-spidev) and the venv is created with --system-site-packages so it
# can see them.
pigpio

# REST API server
fastapi
uvicorn

# Retry helper used by the Volumio client
tenacity

# LCD menu library (forked). Three hard requirements, none of which pip can
# check -- the fork's version did not move for the first two, so an already
# installed copy looks satisfied either way:
#
#   * create_char() and render_frame(), or the level meter dies at runtime with
#     AttributeError rather than failing at install time
#   * the render-performance work (merged in #6): a frame takes ~2.7ms to write
#     instead of ~14ms. The meter defaults to 60fps and offers 120, which the
#     older timing cannot serve -- it would sit at 84% of wall time mid-render
#     and starve the SPI button polling.
#   * 2.4.0 or newer, for the enable-pulse hold and resync_display(). Without
#     it the bus desyncs after an hour or so of the meter and stays broken
#     until the service restarts -- see NOTES.md ("Display driver"). The meter
#     degrades quietly on an older copy rather than crashing, so this one gives
#     no symptom at install time either.
#
# So when updating an existing install, force it: pip will otherwise see the
# installed version and skip the reinstall. That applies to re-testing the same
# branch too -- pushing to it does not move the version number, so pip has no
# way to tell the new commit from the installed one:
#
#   pip install --force-reinstall --no-deps #     git+https://github.com/domb84/rpi-lcd-menu.git@fix-display-corruption
#
# TEMPORARY: pinned to the branch, not master, while the desync fix is being
# tested on the device. Put this back to @master once it is merged.
git+https://github.com/domb84/rpi-lcd-menu.git@fix-display-corruption
